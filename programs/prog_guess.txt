# Інтерактивна гра. Порівняння є лише == та !=, тому "менше" - це відлік
# обох чисел до нуля (додатні числа!).
факт горнятко_32чл SECRET := 7
fn менше := (горнятко_32чл a, горнятко_32чл b) -> заповненість
{
    відкрита горнятко_32чл x := a
    відкрита горнятко_32чл y := b
    допоки (повне)
    {
        якщо (y == 0)
        {
            чайна_перерва пусте
        }
        якщо (x == 0)
        {
            чайна_перерва повне
        }
        x - 1 => x
        y - 1 => y
    }
    чайна_перерва пусте
}
для (відкрита горнятко_32чл спроба := 1; ; спроба + 1 => спроба)
{
    горнятко_32чл g := read()
    якщо (g == SECRET)
    {
        print("вгадав")
        print(спроба)
        чайна_перерва спроба
    }
    інакше
    {
        якщо (менше(g, SECRET) == повне)
        {
            print("більше")
        }
        інакше
        {
            print("менше")
        }
    }
}
чайна_перерва 0
