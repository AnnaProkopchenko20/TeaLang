# Колатц: кількість кроків для числа з вводу. Парність - перемиканням прапорця
# n разів, половина - рахунком по два. Вводити додатні числа!
fn парне := (горнятко_32чл n) -> заповненість
{
    відкрита заповненість p := пусте
    для (відкрита горнятко_32чл i := 0; i != n; i + 1 => i)
    {
        !p => p
    }
    чайна_перерва !p
}
fn половина := (горнятко_32чл n) -> горнятко_32чл
{
    відкрита горнятко_32чл a := 0
    відкрита горнятко_32чл h := 0
    допоки (a != n)
    {
        a + 2 => a
        h + 1 => h
    }
    чайна_перерва h
}
відкрита горнятко_32чл n := read()
відкрита горнятко_32чл steps := 0
допоки (n != 1)
{
    якщо (парне(n) == повне)
    {
        половина(n) => n
    }
    інакше
    {
        n * 3 + 1 => n
    }
    steps + 1 => steps
}
print(steps)
чайна_перерва steps
