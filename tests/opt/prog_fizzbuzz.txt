# FizzBuzz без ділення: два лічильники, що скидаються
відкрита горнятко_32чл c3 := 0
відкрита горнятко_32чл c5 := 0
для (відкрита горнятко_32чл i := 1; i != 16; i + 1 => i)
{
    c3 + 1 => c3
    c5 + 1 => c5
    якщо (c3 == 3)
    {
        якщо (c5 == 5)
        {
            print("FizzBuzz")
            0 => c5
        }
        інакше
        {
            print("Fizz")
        }
        0 => c3
    }
    інакше
    {
        якщо (c5 == 5)
        {
            print("Buzz")
            0 => c5
        }
        інакше
        {
            print(i)
        }
    }
}
