# Прості числа до 30 пробним діленням. Ділення немає, тому "n ділиться на d"
# перевіряється лічильником залишку, що скидається при досягненні d.
fn ділиться := (горнятко_32чл n, горнятко_32чл d) -> заповненість
{
    відкрита горнятко_32чл r := 0
    для (відкрита горнятко_32чл k := 0; k != n; k + 1 => k)
    {
        r + 1 => r
        якщо (r == d)
        {
            0 => r
        }
    }
    заповненість res := r == 0
    чайна_перерва res
}
відкрита горнятко_32чл n := 2
відкрита горнятко_32чл count := 0
відкрита заповненість prime := повне
допоки (n != 31)
{
    повне => prime
    для (відкрита горнятко_32чл d := 2; d != n; d + 1 => d)
    {
        якщо (ділиться(n, d) == повне)
        {
            пусте => prime
        }
    }
    якщо (prime == повне)
    {
        print(n)
        count + 1 => count
    }
    n + 1 => n
}
