# Рекурсивна функція не інлайниться і не видаляється
fn факторіал := (горнятко_32чл n) -> горнятко_32чл
{
    якщо (n == 0)
    {
        чайна_перерва 1
    }
    горнятко_32чл m := факторіал(n - 1)
    горнятко_32чл r := n * m
    чайна_перерва r
}
горнятко_32чл результат := факторіал(5)
print(результат)
чайна_перерва результат
